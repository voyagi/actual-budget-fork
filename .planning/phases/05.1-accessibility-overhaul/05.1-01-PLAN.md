```typescript
function getConsentUrgencyColor(
  validUntil: string | null,
): { text: string; background?: string } | null {
  if (!validUntil) return null;
  const now = new Date();
  const expiry = new Date(validUntil);
  const msUntilExpiry = expiry.getTime() - now.getTime();
  const daysUntilExpiry = msUntilExpiry / (1000 * 60 * 60 * 24);
  if (daysUntilExpiry <= 0) { return { text: theme.errorText }; }
  else if (daysUntilExpiry <= 7) { return { text: theme.warningText }; }
  else if (daysUntilExpiry <= 14) { return { text: theme.noticeText }; }
  return null;
}
```
