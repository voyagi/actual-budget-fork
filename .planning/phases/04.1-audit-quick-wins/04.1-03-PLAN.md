**B. Add localStorage cleanup on mount (dsg-5):**

Add a `useEffect` inside the main `ConsentExpiryBanner` export function (line 274) that runs once on mount to clean up old dismiss keys. Place it after the `useConsentExpiry()` call and before the early return:

```typescript
export function ConsentExpiryBanner() {
  const { sessions } = useConsentExpiry();
  const [, forceUpdate] = React.useReducer(x => x + 1, 0);

  // Clean up old consent dismiss keys from localStorage
  React.useEffect(() => {
    const today = new Date().toDateString();
    const keysToDelete: string[] = [];
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key?.startsWith('consent-dismissed-') && !key.endsWith(today)) {
        keysToDelete.push(key);
      }
    }
    for (const key of keysToDelete) {
      localStorage.removeItem(key);
    }
  }, []);

  // ... rest of component
```

CRITICAL: Use two-pass approach (collect keys first, then delete). Deleting during index iteration corrupts the index and skips keys.
