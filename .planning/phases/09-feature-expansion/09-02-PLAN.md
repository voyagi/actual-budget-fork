    **createTarGz(sourceDir: string, outputPath: string): Promise<void>**
    Helper function to archive a directory into a .tar.gz file using Node.js built-in modules (no `node-tar` dependency).

    Implementation approach: Create a simple tar archive using raw binary writes. The tar format is straightforward for this use case (a flat set of files within subdirectories).

    Alternative simpler approach: Since the backup directory contains only SQLite files and metadata.json, use a simpler strategy:
    - Walk the backup directory to collect all files with their relative paths
    - For each file, write a tar header (512-byte block: name, size, type, checksum) followed by file content padded to 512-byte boundary
    - Pipe the resulting tar stream through `createGzip()` to produce `.tar.gz`

    ```typescript
    async function createTarGz(sourceDir: string, outputPath: string): Promise<void> {
      const files: Array<{ relativePath: string; absolutePath: string; size: number }> = [];
      // Recursively collect all files
      async function collectFiles(dir: string, prefix: string) {
        const entries = await fsp.readdir(dir, { withFileTypes: true });
        for (const entry of entries) {
          const absPath = path.join(dir, entry.name);
          const relPath = prefix ? `${prefix}/${entry.name}` : entry.name;
          if (entry.isDirectory()) {
            await collectFiles(absPath, relPath);
          } else if (entry.isFile()) {
            const stat = await fsp.stat(absPath);
            files.push({ relativePath: relPath, absolutePath: absPath, size: stat.size });
          }
        }
      }
      await collectFiles(sourceDir, '');

      // Write tar.gz: tar header per file + content + padding, then gzip
      const { Readable } = await import('node:stream');
      const tarStream = new Readable({ read() {} });

      for (const file of files) {
        // Create 512-byte tar header
        const header = Buffer.alloc(512);
        // Name (100 bytes)
        header.write(file.relativePath.slice(0, 100), 0, 100, 'utf8');
        // Mode (8 bytes at offset 100)
        header.write('0000644\0', 100, 8, 'utf8');
        // UID/GID (8 bytes each at 108, 116)
        header.write('0001000\0', 108, 8, 'utf8');
        header.write('0001000\0', 116, 8, 'utf8');
        // Size (12 bytes at offset 124, octal)
        header.write(file.size.toString(8).padStart(11, '0') + '\0', 124, 12, 'utf8');
        // Mtime (12 bytes at 136)
        header.write(Math.floor(Date.now() / 1000).toString(8).padStart(11, '0') + '\0', 136, 12, 'utf8');
        // Type flag (1 byte at 156): '0' = regular file
        header.write('0', 156, 1, 'utf8');
        // Checksum placeholder (8 bytes at 148): spaces initially
        header.write('        ', 148, 8, 'utf8');
        // Compute checksum (sum of all bytes treating checksum field as spaces)
        let checksum = 0;
        for (let i = 0; i < 512; i++) checksum += header[i];
        header.write(checksum.toString(8).padStart(6, '0') + '\0 ', 148, 8, 'utf8');

        tarStream.push(header);
        // File content
        const content = await fsp.readFile(file.absolutePath);
        tarStream.push(content);
        // Pad to 512-byte boundary
        const remainder = content.length % 512;
        if (remainder > 0) tarStream.push(Buffer.alloc(512 - remainder));
      }
      // End-of-archive marker (two 512-byte zero blocks)
      tarStream.push(Buffer.alloc(1024));
      tarStream.push(null); // end stream

      const gzip = createGzip();
      const output = createWriteStream(outputPath);
      await pipeline(tarStream, gzip, output);
    }
    ```

    if (process.env.ENABLE_AUTO_BACKUP !== 'false') {
      const backupSchedule = process.env.BACKUP_CRON_SCHEDULE ?? '0 2 * * *';
      cron.schedule(backupSchedule, () => {
        runBackup().then(result => {
          if (result.success) {
            logger.info('Backup completed', {

            logger.error('Backup failed', { error: result.error });
            recordBackupRun(0, false);
            triggerAlert({
              event_type: 'backup_failure',
              message: `Daily backup failed: ${result.error}`,
              severity: 'error',
            }).catch(() => {});
          }
        }).catch(err => {
          logger.error('Backup cron unhandled error', {
            error: err instanceof Error ? err.message : String(err),
          });
          recordBackupRun(0, false);
          triggerAlert({
            event_type: 'backup_failure',
            message: `Daily backup failed: ${err instanceof Error ? err.message : String(err)}`,
            severity: 'error',
          }).catch(() => {});
        });
      });
