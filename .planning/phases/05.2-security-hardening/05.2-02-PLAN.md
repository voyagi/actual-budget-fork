       export function isValidRedirectUrl(url) {
         const serverHostname = getServerHostname();

         if (!serverHostname) {
           return false;
         }

         try {
           const redirectUrl = new URL(url);
           const serverUrl = new URL(serverHostname);

           const hostnamesMatch = redirectUrl.hostname === serverUrl.hostname;
           const portsMatch = redirectUrl.port === serverUrl.port;
           const schemesMatch = redirectUrl.protocol === serverUrl.protocol;

           if (hostnamesMatch && portsMatch && schemesMatch) {
             return true;
           }

           console.warn(
             `[openid] Blocked redirect URL: ${url} (server: ${serverHostname})`,
           );
           return false;
         } catch {
           return false;
         }
       }

       Update `regexHelper` to handle this:
       ```typescript
       function regexHelper(
         mapRegex: (regex: string, flags: string) => string | RegExp,
         mapNonRegex: (value: string) => string | RegExp,
         apply: (value: string, regex: string | RegExp, replace: string) => string,
       ) {
         return (value: unknown, regex: unknown, replace: unknown) => {
           if (value == null) {
             return null;
           }

           if (typeof regex !== 'string' || typeof replace !== 'string') {
             return '';
           }

           const match = regexTest.exec(regex);
           // Regex-literal branch: /regex/flags - use timeout protection
           if (match) {
             const regexp = mapRegex(match[1], match[2]);
             if (regexp instanceof RegExp) {
               // Determine method from the apply function
               const applyFn = apply.toString().includes('replaceAll')
                 ? 'replaceAll' as const
                 : 'replace' as const;
               return runRegexWithTimeout(
                 String(value), regexp, String(replace), applyFn,
               );
             }
             // String fallback (shouldn't happen but type-safe)
             try {
               return apply(String(value), regexp, String(replace));
             } catch (e) {
               logger.log(
                 `[handlebars] Regex operation failed: ${(e as Error).message}`,
               );
               return String(value);
             }
           }

           // Non-regex branch: escaped input, safe, runs synchronously
           const regexp = mapNonRegex(String(regex));
           try {
             return apply(String(value), regexp, String(replace));
           } catch (e) {
             logger.log(
               `[handlebars] Regex operation failed: ${(e as Error).message}`,
             );
             return String(value);
           }
         };
       }
       ```

       **Important async consideration:** Handlebars helpers that return Promises render as `[object Promise]` unless the caller awaits them. The executor MUST check how Handlebars templates are evaluated in the rules engine. Search for `Handlebars.compile` or `template(data)` calls in the rules code.

       - If Handlebars is called synchronously: the executor must use the **synchronous fallback** instead of `runRegexWithTimeout`. The synchronous fallback wraps the `apply` call in a try/catch only, which catches runtime errors but cannot interrupt infinite backtracking. In this case, add a comment documenting the limitation and consider adding a regex complexity pre-check (reject patterns with nested quantifiers like `(a+)+` before execution).

       The executor should determine the correct approach and document the choice.
